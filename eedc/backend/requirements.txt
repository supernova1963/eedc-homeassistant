# Ausgeliefert wird `requirements.lock` — die daraus aufgelöste, gepinnte Liste
# (uv pip compile, Python 3.11, plattformübergreifend für amd64 und aarch64). Beide

#   cd eedc/backend && uv pip compile --python-version 3.11 --universal requirements.txt -o requirements.lock
